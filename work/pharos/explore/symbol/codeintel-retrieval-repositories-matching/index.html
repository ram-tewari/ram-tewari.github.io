def repositories_matching(db: Session, ref: str | None) -> list[Repository]:    """Resolve a repository reference: id, canonical URL, name or URL suffix."""    if not ref:        return []    ref = ref.strip()    try:        rid = uuid.UUID(ref)        repo = db.get(Repository, rid)        return [repo] if repo else []    except ValueError:        pass    norm = ref.lower().removesuffix(".git").rstrip("/")    repos = db.execute(select(Repository)).scalars().all()    exact = [r for r in repos if r.url.lower() == norm or r.name.lower() == norm]    if exact:        return exact    return [r for r in repos if r.url.lower().endswith("/" + norm) or r.name.lower().endswith("/" + norm)]
