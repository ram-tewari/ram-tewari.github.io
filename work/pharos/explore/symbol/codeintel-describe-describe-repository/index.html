"""    stats = DescribeStats()    todo = _candidates(db, repository_id, only_undocumented, limit)    stats.candidates = len(todo)    files: dict[str, list[str]] = {}    model_name = getattr(llm, "model", None) or "codex"    for start in range(0, len(todo), BATCH):        batch = todo[start:start + BATCH]        issued: dict[str, tuple[DocumentChunk, str, str]] = {}        for i, (chunk, sha) in enumerate(batch, start=1):            path = (chunk.chunk_metadata or {}).get("path", "")            if path not in files:                try:                    files[path] = (checkout / path).read_text(encoding="utf-8", errors="replace").splitlines()                except OSError:                    files[path] = []            code = "\n".join(files[path][(chunk.start_line or 1) -
