Reciprocal rank fusion

Each list contributes 1/(60 + rank). Dense and keyword modes keep their own scores.

    def _fuse(self, mode: str, keyword, dense, terms: list[str], query: str,              hints: list[str] | None = None) -> dict[str, Hit]:        hits: dict[str, Hit] = {}        def hit(cid: str) -> Hit:            if cid not in hits:                hits[cid] = Hit(chunk_id=cid, score=0.0, evidence=Evidence())            return hits[cid]        if mode == "keyword":            for rank, (cid, score) in enumerate(keyword, start=1):                h = hit(cid)                h.score = score                h.evidence.keyword_rank, h.evidence.keyword_score = rank, score            return hits        if mode == "dense":            for rank, (cid, sim) in enumerate(dense, start=1):                h = hit(cid)                h.score = sim                h.evidence.dense_rank, h.evidence.dense_similarity = rank, sim            return hits        for rank, (cid, score) in enumerate(keyword, start=1):            h = hit(cid)            h.score += 1.0 / (RRF_K + rank)            h.evidence.keyword_rank, h.evidence.keyword_score = rank, score        for rank, (cid, sim) in enumerate(dense, start=1):            h = hit(cid)            h.score += 1.0 / (RRF_K + rank)            h.evidence.dense_rank, h.evidence.dense_similarity = rank, sim        # Exact symbol-name match: the query names the symbol directly        # (e.g. "verify_password" or "TokenSigner.verify").        if hits:            self._symbol_boost(hits, terms, query, hints)        return hits
