Down-rank tests and docs

Tests, docs and examples are multiplied by 0.7 after fusion, unless the query asks for them.

PINNED SOURCE Copied from the repository at this commit. Source, not an execution.

    def _finalize(self, hits: dict[str, Hit], k: int, wanted: set[str] = frozenset()) -> list[Hit]:        if not hits:            return []        # Down-rank tests, docs and examples (they repeat the words of the        # code they exercise or describe and crowd out implementations);        # applied after fusion so they still surface when they are the only        # match, and not at all for roles the query asks for.        paths = dict(            self.db.execute(                select(DocumentChunk.id, DocumentChunk.chunk_metadata["path"].as_string()).where(                    DocumentChunk.id.in_([uuid.UUID(c) for c in hits])                )            ).all()        )        for cid, h in hits.items():            role = path_role(paths.get(uuid.UUID(cid)) or "")            if role != "source" and role not in wanted:                h.score *= NOISE_PATH_FACTOR                h.evidence.boosts.append(f"{role}_path_penalty")        ranked = sorted(hits.values(), key=lambda h: (-h.score, h.chunk_id))        return ranked[:k]
