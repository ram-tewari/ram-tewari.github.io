What gets embedded

A chunk is embedded from its signature, docstring, calls and first body lines, not from raw source.

def semantic_summary(sym: Symbol, *, language: str, path: str, module_name: str, body: str = "",                     description: str = "") -> str:    '''The text that gets embedded for a chunk (not the raw source).    Format::        [python] pkg/auth.py :: TokenStore.issue        def issue(self, user):            """Issue a signed token."""            payload = {"sub": user.id}            return self._sign(encode(payload))            calls: [_sign, encode]    ``body`` (see :func:`body_excerpt`) goes before the call list: embedders    truncate long inputs (MiniLM at 256 tokens), and the body says more.    '''    local_name = sym.qualified_name    if module_name and local_name.startswith(module_name + "."):        local_name = local_name[len(module_name) + 1 :]    if sym.kind in ("module", "block"):        header = f"[{language}] {path}"    else:        header = f"[{language}] {path} :: {local_name}"    lines = [header, sym.signature] if sym.signature else [header]    if sym.docstring:        lines.append(f'    """{sym.docstring}"""')    if description:        lines.append(f"    purpose: {description}")    if body:        lines.append(body)    if sym.calls:        lines.append(f"    calls: [{', '.join(sym.calls[:20])}]")    return "\n".join(lines)
