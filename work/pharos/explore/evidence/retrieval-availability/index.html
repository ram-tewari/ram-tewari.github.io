Is dense possible here?

Vectors from different models are never compared. A model mismatch makes dense unavailable.

PINNED SOURCE Copied from the repository at this commit. Source, not an execution.

PHAROS at 975a38dMIT license

    def _dense_unavailable_reason(self, filters: Filters) -> str | None:        if self.embedder is None:            return "no embedding backend configured (EMBEDDING_BACKEND=none)"        stmt = select(DocumentChunk.id).where(            DocumentChunk.embedding_model == self.embedder.model_id,            *self._filter_clauses(filters),        ).limit(1)        if self.db.execute(stmt).first() is not None:            return None        models = self.db.execute(            select(DocumentChunk.embedding_model)            .where(DocumentChunk.embedding_model.isnot(None), *self._filter_clauses(filters))            .distinct()            .limit(5)        ).scalars().all()        if models:            return (                f"index was embedded with {sorted(models)}, query embedder is "                f"{self.embedder.model_id!r}; re-index with the same EMBEDDING_BACKEND"            )        return "no chunk embeddings in the index"
