Pick a parser

Python uses the stdlib ast module, seven other languages use tree-sitter, and the rest fall back to line windows that are recorded as fallback.

def parse_file(path: str, source: str) -> ParseResult | None:    """Parse ``source`` (the content of repository-relative ``path``).    Returns None for files that are not indexable source code.    """    spec = detect_language(path)    if spec is None:        return None    if spec.parser == "python-ast":        result = parse_python(path, source)        return result if result is not None else line_chunks(path, source, spec.name, "syntax_error")    if spec.parser == "tree-sitter":        try:            result = parse_treesitter(path, source, spec.name, spec.grammar, _module_path(path))        except ImportError as exc:            return line_chunks(path, source, spec.name, str(exc))        if len(result.symbols) <= 1 and result.partial:            # Only the module symbol and the tree is broken: line chunks            # retrieve better than one opaque whole-file chunk.            return line_chunks(path, source, spec.name, "parse_error")        return result    return line_chunks(path, source, spec.name, "unsupported_language")
