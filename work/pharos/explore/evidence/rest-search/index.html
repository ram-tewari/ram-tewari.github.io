The REST search endpoint

Reports the mode it actually used, its candidates and its timings.

@router.post("/search/code", response_model=SearchResponse)def search_code(body: SearchRequest, db: Session = Depends(get_sync_db)):    """Chunk-level retrieval. ``mode=auto`` reports what it actually used."""    filters = _filters(db, body.repository, body.languages, body.kinds, body.path_prefix)    try:        outcome = Retriever(db, get_embedder()).search(body.query, mode=body.mode, k=body.k, filters=filters,                                                       hints=body.hints)    except ModeUnavailable as exc:        raise _mode_unavailable(exc) from exc    return SearchResponse(        query=outcome.query,        mode_requested=outcome.mode_requested,        mode_used=outcome.mode_used,        embedding_model=outcome.embedding_model,        warnings=outcome.warnings,        results=[hit_to_dict(h, i) for i, h in enumerate(outcome.hits, start=1)],        timings_ms=outcome.timings_ms,        candidates=outcome.candidates,    )
