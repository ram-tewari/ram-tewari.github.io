Pointer to request

The blob SHA comes from the row written at index time, not from the file being checked.

def source_requests(db: Session, hits: list[Hit]) -> list[SourceRequest]:    if not hits:        return []    repo_rows = {        str(r.id): r        for r in db.execute(            select(Repository).where(Repository.id.in_({uuid.UUID(h.repository_id) for h in hits}))        ).scalars()    }    blob = dict(        db.execute(            select(Resource.id, Resource.content_sha).where(                Resource.id.in_({uuid.UUID(h.resource_id) for h in hits if h.resource_id})            )        ).all()    )    reqs = []    for h in hits:        repo = repo_rows[h.repository_id]        start, end = h.start_line or 1, h.end_line or 1        if h.kind == "module":            end = min(end, start + MODULE_CODE_LINES - 1)        reqs.append(            SourceRequest(                key=h.chunk_id,                source_type=repo.source_type,                repository_url=repo.url,                local_path=repo.local_path,                commit=h.commit_sha or "",                path=h.path,                start_line=start,                end_line=end,                blob_sha=blob.get(uuid.UUID(h.resource_id)) if h.resource_id else None,            )        )    return reqs
